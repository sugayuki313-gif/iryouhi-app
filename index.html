/* ========================================================= 修正版：高額療養費計算（厚生労働省ガイドライン準拠） ========================================================= */ function calculateCorrectCap(ageCategory, category, cost, treatmentType, multiple) { let cap = 0; let multipleCap = null; if (ageCategory === "under70") { switch (category) { case "top": // 区分ア（年収約1,160万円〜） cap = 252600 + Math.max(0, cost - 842000) * 0.01; multipleCap = 140100; break; case "high": // 区分イ（年収約770万〜1,160万円） cap = 167400 + Math.max(0, cost - 558000) * 0.01; multipleCap = 93000; break; case "middle": // 区分ウ（年収約370万〜770万円） cap = 80100 + Math.max(0, cost - 267000) * 0.01; multipleCap = 44400; break; case "low": // 区分エ（年収〜約370万円） cap = 57600; multipleCap = 44400; break; case "nonTax": // 区分オ（住民税非課税世帯） cap = 35400; multipleCap = 24600; break; } } else { // 70歳以上の計算（外来個人上限／世帯上限の分岐） if (category === "high") { // 現役並み所得（3割） cap = 80100 + Math.max(0, cost - 267000) * 0.01; // 一般的な現役並みⅢの例 multipleCap = 44400; } else if (category === "middle") { // 一般（1割・2割） if (treatmentType === "outpatient") { cap = 18000; // 外来個人上限 } else { cap = 57600; // 世帯上限 multipleCap = 44400; } } else if (category === "low") { // 低所得Ⅱ（非課税） cap = (treatmentType === "outpatient") ? 8000 : 24600; } else if (category === "nonTax") { // 低所得Ⅰ cap = (treatmentType === "outpatient") ? 8000 : 15000; } } if (multiple === "3" && multipleCap !== null) { cap = Math.min(cap, multipleCap); } return cap; }
