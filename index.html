function calculateCorrectCap(ageCategory, category, cost, treatmentType, multiple) { let cap = 0; let multipleCap = null; if (ageCategory === "under70") { // 69歳以下の計算式 switch (category) { case "top": // 区分ア（年収約1,160万円〜） cap = 252600 + Math.max(0, cost - 842000) * 0.01; multipleCap = 140100; break; case "high": // 区分イ（年収約770万〜1,160万円） cap = 167400 + Math.max(0, cost - 558000) * 0.01; multipleCap = 93000; break; case "middle": // 区分ウ（年収約370万〜770万円） cap = 80100 + Math.max(0, cost - 267000) * 0.01; multipleCap = 44400; break; case "low": // 区分エ（年収〜約370万円） cap = 57600; multipleCap = 44400; break; case "nonTax": // 区分オ（住民税非課税者） cap = 35400; multipleCap = 24600; break; } } else { // 70歳以上の計算式 switch (category) { case "top": // 現役並みⅢ（年収約1,160万円〜） cap = 252600 + Math.max(0, cost - 842000) * 0.01; multipleCap = 140100; break; case "high": // 現役並みⅡ（年収約770万〜1,160万円） cap = 167400 + Math.max(0, cost - 558000) * 0.01; multipleCap = 93000; break; case "middle_high": // 現役並みⅠ（年収約370万〜770万円） cap = 80100 + Math.max(0, cost - 267000) * 0.01; multipleCap = 44400; break; case "middle": // 一般（年収156万〜約370万円） if (treatmentType === "outpatient") { cap = 18000; // 個人単位の外来上限 } else { cap = 57600; // 世帯単位の上限 multipleCap = 44400; } break; case "low": // 低所得Ⅱ（住民税非課税世帯） cap = (treatmentType === "outpatient") ? 8000 : 24600; break; case "nonTax": // 低所得Ⅰ（判定基準を満たす非課税） cap = (treatmentType === "outpatient") ? 8000 : 15000; break; } } // 多数回該当の判定処理 if (multiple === "3" && multipleCap !== null) { cap = Math.min(cap, multipleCap); } return Math.floor(cap); }
